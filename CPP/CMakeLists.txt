cmake_minimum_required(VERSION 3.16)
project(mixer_slice VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(STRAW_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../hic-infra/straw/C++" CACHE PATH
    "Path to the straw C++ source directory (with straw.h and straw.cpp)")
if(NOT EXISTS "${STRAW_SOURCE_DIR}/straw.h")
  message(FATAL_ERROR "Set -DSTRAW_SOURCE_DIR=/path/to/straw/C++")
endif()
find_package(CURL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(zstd CONFIG QUIET)
if(TARGET zstd::libzstd_static)
  set(SLICE_ZSTD zstd::libzstd_static)
elseif(TARGET zstd::libzstd_shared)
  set(SLICE_ZSTD zstd::libzstd_shared)
elseif(TARGET zstd::libzstd)
  set(SLICE_ZSTD zstd::libzstd)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
  set(SLICE_ZSTD PkgConfig::ZSTD)
endif()
# Build just the native reader library, without straw's CLI, C API, or tests.
add_library(slice_straw STATIC "${STRAW_SOURCE_DIR}/straw.cpp"
  "${STRAW_SOURCE_DIR}/straw_v10.cpp" "${STRAW_SOURCE_DIR}/hbs.cpp"
  "${STRAW_SOURCE_DIR}/norm_vectors.cpp")
set_target_properties(slice_straw PROPERTIES CXX_STANDARD 14)
target_include_directories(slice_straw PUBLIC "${STRAW_SOURCE_DIR}")
target_link_libraries(slice_straw PUBLIC CURL::libcurl ZLIB::ZLIB Threads::Threads ${SLICE_ZSTD})
add_library(slice_core src/numerics.cpp src/reader.cpp src/pipeline.cpp src/projection.cpp src/translocations.cpp)
target_include_directories(slice_core PUBLIC include)
target_link_libraries(slice_core PUBLIC slice_straw)
target_compile_options(slice_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(slice src/main.cpp)
target_link_libraries(slice PRIVATE slice_core)
install(TARGETS slice RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(slice_numerics_test tests/numerics.cpp)
  target_link_libraries(slice_numerics_test PRIVATE slice_core)
  add_test(NAME numerics COMMAND slice_numerics_test)
  set_tests_properties(numerics PROPERTIES TIMEOUT 30)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_package(Java COMPONENTS Runtime Development QUIET)
  if(Java_FOUND)
    add_test(NAME java_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/java_parity.py" "$<TARGET_FILE:slice>")
    set_tests_properties(java_parity PROPERTIES TIMEOUT 240)
  endif()
  add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:slice>")
  set_tests_properties(integration PROPERTIES TIMEOUT 180)
endif()

if(BUILD_TESTING)
  add_executable(adaptive_test tests/adaptive.cpp)
  target_link_libraries(adaptive_test PRIVATE Threads::Threads)
  target_include_directories(adaptive_test PRIVATE include/slice)
  add_test(NAME adaptive COMMAND adaptive_test)
endif()

if(BUILD_TESTING)
  add_executable(bed_output_test tests/bed_output.cpp)
  target_include_directories(bed_output_test PRIVATE include)
  add_test(NAME bed_output COMMAND bed_output_test "${CMAKE_CURRENT_BINARY_DIR}/bed_output_test.bed")
endif()

if(BUILD_TESTING)
  add_executable(cis_coverage_test tests/cis_coverage.cpp)
  target_include_directories(cis_coverage_test PRIVATE include)
  add_test(NAME cis_coverage COMMAND cis_coverage_test)
endif()

if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME cis_coverage_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cis_coverage_integration.py" $<TARGET_FILE:slice>)
endif()
